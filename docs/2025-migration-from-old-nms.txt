-- CONTRACT VERSIONS
INSERT INTO contract_versions (
    id,
    contract_id,

    valid_from,
    valid_until,
    obligation_until,
    conclusion_date,

    number_of_amendments,
    note,

    created,
    modified,

    created_by,
    modified_by
)
SELECT
    uuid_generate_v4()          AS id,
    c.id                        AS contract_id,

    CASE
        WHEN ip."billingFrom" IS NOT NULL
            THEN ip."billingFrom"
        WHEN EXTRACT(DAY FROM ip."installationDate") <= 10
            THEN date_trunc('month', ip."installationDate")::date
        ELSE
            (date_trunc('month', ip."installationDate") + INTERVAL '1 month')::date
    END                         AS valid_from,

    c.termination_date          AS valid_until,
    ip.bond                     AS obligation_until,
    NULL                        AS conclusion_date,

    0                           AS number_of_amendments,
    NULL                        AS note,

    NULL                        AS created,
    NULL                        AS modified,

    NULL                        AS created_by,
    NULL                        AS modified_by
FROM contracts c
JOIN old.ip ip
    ON ip.contract_uuid = c.id
WHERE ip."installationDate" IS NOT NULL
   OR ip."billingFrom" IS NOT NULL;

INSERT INTO public.billings (
    service_id,
    text,
    price,
    billing_from,
    billing_until,
    note,
    customer_id,
    contract_id,
    quantity,
    separate_invoice,
    created,
    modified
)
SELECT
    -- service_id (if exists)
    b.service_uuid,

    -- text (only overrides)
    CASE
        WHEN b.service_uuid IS NOT NULL THEN NULL
        ELSE b.text
    END,

    -- price (NULL = take from pricelist)
    CASE
        WHEN b.service_uuid IS NOT NULL
             AND s.price IS NOT NULL
             AND b.cost = s.price
        THEN NULL
        ELSE b.cost
    END,

    -- billing period
    b."billingFrom",
    b."billingTill",

    -- note
    b.note,

    -- customer_id from old.users
    u.uuid AS customer_id,

    -- contract_id (temporary null)
    NULL AS contract_id,

    -- defaults
    1     AS quantity,
    false AS separate_invoice,

    -- timestamps
    b.insertDate AS created,
    b.changeDate AS modified

FROM old.billing b
JOIN old.users u
    ON u."userId" = b."userId"
LEFT JOIN services s
    ON s.id = b.service_uuid
WHERE b.active = 1;
